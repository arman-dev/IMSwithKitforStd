@workspace_view — এক decorator, চার সিদ্ধান্ত

apps/core/permissions.py · ৫৭টি ভিউতে ব্যবহৃত
def workspace_view(*allowed_roles, allow_platform_admin=True):
    def decorator(view_func):
        @wraps(view_func)
        def _wrapped(request, slug, *args, **kwargs):
            if not request.user.is_authenticated:
                return redirect(f"/accounts/login/?next={request.path}")

            workspace = Workspace.objects.filter(
                slug=slug, is_active=True).first()
            if workspace is None:
                raise Http404("Workspace not found.")

            membership = get_membership(request.user, workspace)
            if membership is None:
                # প্ল্যাটফর্ম অ্যাডমিনও অন্যের ব্যক্তিগত
                # রেকর্ড দেখতে পারে না
                raise Http404("Workspace not found.")

            if allowed_roles and membership.role not in allowed_roles:
                raise PermissionDenied(
                    "Your role in this workspace does not allow this action.")

            request.workspace = workspace
            request.membership = membership
            request.session["active_workspace_id"] = workspace.pk
            return view_func(request, slug, *args, **kwargs)
        return _wrapped
    return decorator
