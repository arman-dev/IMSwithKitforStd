transaction.atomic() — অর্ধেক অ্যাকাউন্ট কখনো নয়
def form_valid(self, form):
    with transaction.atomic():
        user = form.save()
        workspace = Workspace.objects.create(
            kind=WorkspaceKinds.PERSONAL,
            name=form.cleaned_data["workspace_name"],
            owner=user)
        Membership.objects.create(
            user=user, workspace=workspace,
            role=Roles.TEACHER)
    login(self.request, user)
    self.request.session[SESSION_KEY] = workspace.pk
    return redirect("reports:dashboard", slug=workspace.slug)
