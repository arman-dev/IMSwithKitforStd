set -euo pipefail
STAMP=$(date +%Y%m%d-%H%M%S)
DEST="${BACKUP_DIR:-./backups}"

pg_dump -Fc -f "$DEST/ims-$STAMP.dump"          # ডেটাবেস

    2>/dev/null || true                          # লোগো/ফাইল

find "$DEST" -name '*.dump'   -mtime +14 -delete
find "$DEST" -name '*.tar.gz' -mtime +14 -delete

# cron: রাত ২টায়
