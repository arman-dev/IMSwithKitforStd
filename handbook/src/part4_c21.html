৯টি এক্সটেনশন · ১০ MB · clean()-এ যাচাই
class Material(TimeStamped):
    ALLOWED_EXTENSIONS = {"pdf", "docx", "xlsx", "pptx", "txt",
                          "jpg", "jpeg", "png", "zip"}
    MAX_BYTES = 10 * 1024 * 1024

    def clean(self):
        ext = (self.file.name.rsplit(".", 1)[-1].lower()
               if self.file else "")
        if ext and ext not in self.ALLOWED_EXTENSIONS:
            raise ValidationError(
                {"file": f"File type .{ext} is not allowed."})
        if self.file and hasattr(self.file, "size") \
                and self.file.size > self.MAX_BYTES:
            raise ValidationError(
                {"file": "File is larger than 10 MB."})

MaterialForm.save(commit=False) → obj.clean() → obj.save()
@workspace_view(*TEACHING)
def material_create(request, slug):
    form = MaterialForm(request.POST or None,
                        request.FILES or None,
                        workspace=request.workspace)
    if request.method == "POST" and form.is_valid():
        obj = form.save(commit=False)
        obj.workspace = request.workspace       # স্কোপ ভিউ থেকে
        obj.uploaded_by = request.user
        obj.clean()                             # আবার যাচাই
        obj.save()
        messages.success(request, "Material uploaded.")
        return redirect("lessons:material_list", slug=slug)

@workspace_view(*ALL_ROLES)
def material_download(request, slug, pk):
    """Files are never served from a public URL."""
    material = get_object_or_404(Material, pk=pk,
                                  workspace=request.workspace)
    allowed = request.membership.role in (
        "ORG_ADMIN", "COORDINATOR", "TEACHER") or (
        teacher_can_access_class(request.user, material.class_group))
    student = getattr(request.user, "student_profile", None)
    if student:
        allowed = material.is_published and student.enrollments.filter(
            class_group=material.class_group, is_active=True).exists()
    if not allowed:
        raise PermissionDenied("You cannot download this file.")
    try:
        return FileResponse(material.file.open("rb"),
                            as_attachment=True,
                            filename=material.file.name.split("/")[-1])
    except FileNotFoundError:
        raise Http404("The file is missing on disk.")

from django.core.files.uploadedfile import SimpleUploadedFile

def test_rejects_exe_and_accepts_pdf(self):
    self.client.force_login(self.teacher)
    url = reverse("lessons:material_create",
                  kwargs={"slug": self.ws.slug})

    bad = SimpleUploadedFile("notes.exe", b"MZ\x90\x00",
                             content_type="application/pdf")
    self.client.post(url, {"title": "X", "class_group": self.cls.pk,
                           "file": bad})
    self.assertFalse(Material.objects.exists())

    good = SimpleUploadedFile("notes.pdf", b"%PDF-1.4 ok",
                              content_type="application/pdf")
    self.client.post(url, {"title": "X", "class_group": self.cls.pk,
                           "file": good})
