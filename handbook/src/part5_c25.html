def visible_classes(user, workspace, membership=None):
    membership = membership or membership_for(user, workspace)
    qs = ClassGroup.objects.filter(workspace=workspace,
                                   is_archived=False)
    if membership is None:
        return qs.none()

    if membership.role == Roles.ORG_ADMIN:
        return qs

    if membership.role == Roles.COORDINATOR:
        scopes = CoordinatorScope.objects.filter(
            workspace=workspace, user=user, is_active=True)
        dept_ids = list(scopes.exclude(department=None)
                        .values_list("department_id", flat=True))
        class_ids = list(scopes.exclude(class_group=None)
                         .values_list("class_group_id", flat=True))
        return qs.filter(models_q(dept_ids, class_ids))

    if membership.role == Roles.TEACHER:
        return qs.filter(assignments__teacher=user,
                         assignments__is_active=True).distinct()

    if membership.role == Roles.STUDENT:
        student = getattr(user, "student_profile", None)
        if student is None:
            return qs.none()
        return qs.filter(enrollments__student=student,
                         enrollments__is_active=True).distinct()
    return qs.none()

def test_second_active_enrolment_is_blocked_by_the_database(self):
    from django.db import IntegrityError, transaction
    Enrollment.objects.create(workspace=self.ws, student=self.student,
                              class_group=self.cls, is_active=True)
    with self.assertRaises(IntegrityError):
        with transaction.atomic():          # না থাকলে টেস্ট ভেঙে যায়
            Enrollment.objects.create(workspace=self.ws,
                                      student=self.student,
                                      class_group=self.cls,
                                      is_active=True)

def test_ending_an_enrolment_keeps_history(self):
    e.is_active = False; e.end_reason = "Transferred"; e.save()
    e.refresh_from_db()
    self.assertIsNotNone(e.ended_on)          # save() নিজে তারিখ বসায়
    self.assertEqual(Enrollment.objects.filter(student=...).count(), 1)
assertRaises + atomic

IntegrityError-এর পর ট্রানজ্যাকশন ভাঙা থাকে — atomic ছাড়া পরের কুয়েরি ব্যর্থ।

ইতিহাস ≠ মুছে ফেলা

ভর্তি বন্ধ মানে সারি থাকবে, ended_on বসবে — সারি সংখ্যা ১-ই থাকে।
