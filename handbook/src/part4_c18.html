একটি field-level নিয়ম, তিনটি সূক্ষ্মতা

clean_admission_no() — ইউনিক, কিন্তু সম্পাদনার সময় নিজেকে বাদ দিয়ে
def clean_admission_no(self):
    value = self.cleaned_data["admission_no"].strip()
    qs = StudentProfile.objects.filter(
        workspace=self.workspace,
        admission_no__iexact=value)
    if self.instance.pk:          # সম্পাদনা করছি
        qs = qs.exclude(pk=self.instance.pk)
    if qs.exists():
        raise forms.ValidationError(
            "Another student in this workspace "
            "already uses this admission number.")
    return value
সূক্ষ্মতানা করলে
workspace= ফিল্টারঅন্য প্রতিষ্ঠানের নম্বরও "দখল" দেখাবে
__iexactA-12 আর a-12 দুইবার ঢুকবে
.strip()শেষে স্পেস থাকলে ভিন্ন মান
exclude(pk=…)নিজের রেকর্ড সম্পাদনা করতে গেলে "আগেই আছে" এরর
return valueপরের ধাপে None পাবে — ডেটা হারাবে
শেষ সারিটি সবচেয়ে বেশি ভুল হয়: clean_<field> কখনো None ফেরাতেই হবে না — পরিষ্কার করা মান ফেরাতে হয়।
শিক্ষকের নোট: "সম্পাদনা করুন" আর "নতুন যোগ করুন" — একই ফর্ম। ছাত্ররা exclude(pk) ভুলে যায়, তারপর বলে "আমার ফর্মটা কাজ করে না।" এখানেই সেই বাগের জন্ম।

# apps/academics/forms.py
class BootstrapFormMixin:
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        for name, field in self.fields.items():
            w = field.widget
            if isinstance(w, forms.CheckboxInput):
                w.attrs.setdefault("class", "form-check-input")
            elif isinstance(w, forms.Select):
                w.attrs.setdefault("class", "form-select")
            else:
                w.attrs.setdefault("class", "form-control")

        help_text="Only active students not already in "

    def __init__(self, *args, workspace=None,
                 class_group=None, **kwargs):
        super().__init__(*args, **kwargs)
        qs = StudentProfile.objects.filter(
            workspace=workspace,
            status=StudentProfile.Status.ACTIVE)
        if class_group is not None:

                class_group=class_group, is_active=True,

        """Read the per-student radio values."""

    if logo and hasattr(logo, "size"):
        if logo.size > 2 * 1024 * 1024:
            raise forms.ValidationError(
                "Logo must be under 2 MB.")
        ext = logo.name.rsplit(".", 1)[-1].lower()
        if ext not in ("png", "jpg", "jpeg", "svg", "webp"):
            raise forms.ValidationError(
                "Use PNG, JPG, SVG or WebP.")
    return logo
ঝুঁকি ১: আকার

৫০ MB লোগো আপলোড হলে ডিস্ক ও ব্যাকআপ ফুলে যায়।

ঝুঁকি ২: ধরন

HTML/SVG স্ক্রিপ্টসহ আপলোড হলে XSS-এর দরজা খোলে — SVG-তে JS থাকতে পারে।

ঝুঁকি ৩: নাম

ব্যবহারকারীর দেওয়া ফাইলনাম সরাসরি ব্যবহার করবেন না — Django নিজেই নাম বদলায়।
