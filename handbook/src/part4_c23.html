@property
def ends_on(self):
    return (self.trial_ends_on if self.status == self.Status.TRIAL
            else self.current_period_end)

@property
def days_left(self):
    end = self.ends_on
    return None if end is None else (end - timezone.localdate()).days

@property
def state_label(self):
    if self.is_writable:
        return "Trial" if self.status == self.Status.TRIAL else "Active"
    return ("Expired" if self.status in (self.Status.TRIAL, self.Status.ACTIVE)
            else self.get_status_display())

@transaction.atomic
def activate(workspace, plan, *, days=None, amount=None,
             method=Invoice.Method.BKASH, reference="", actor=None):
    sub = ensure_subscription(workspace)
    today = timezone.localdate()
    start = today
    if (sub.current_period_end and sub.current_period_end > today
            and sub.plan_id == plan.pk):
        start = sub.current_period_end      # আগেভাগে নবায়ন: দিন হারায় না
    end = start + timedelta(days=days) if days else period_end(start, plan)

    sub.plan, sub.status = plan, Subscription.Status.ACTIVE
    sub.current_period_start, sub.current_period_end = start, end
    sub.save()

    return Invoice.objects.create(
        workspace=workspace, subscription=sub, plan=plan,
        period_start=start, period_end=end,
        amount=amount if amount is not None else plan.price,
        status=Invoice.Status.PAID, method=method,
        reference=reference[:80], paid_on=today, recorded_by=actor)

def save(self, *args, **kwargs):
    if not self.number:
        self.number = self._next_number()

@staticmethod
def _next_number() -> str:
    year = timezone.localdate().year
    prefix = f"{settings.INVOICE_PREFIX}-{year}-"
    last = (Invoice.objects.filter(number__startswith=prefix)
            .order_by("-number").values_list("number", flat=True).first())
    n = int(last.split("-")[-1]) + 1 if last else 1
    return f"{prefix}{n:04d}"

def seat_block_reason(workspace) -> str | None:
    sub = subscription_for(workspace)
    if sub is None:
        return None                 # ব্যক্তিগত ওয়ার্কস্পেস
    allowed = sub.seats_allowed     # seats_override or plan.max_students
    if allowed is None:
        return None                 # অসীম
    if sub.used_students() >= allowed:
        return (f"Your plan allows {allowed} active students. "
                f"Upgrade to add more.")

class CustomDomainMiddleware:
    def __call__(self, request):
        request.domain_workspace = None
        host = request.get_host().split(":")[0].lower()
        branding = (WorkspaceBranding.objects
                    .select_related("workspace")
                    .filter(custom_domain__iexact=host,
                            domain_verified=True,
                            workspace__is_active=True)
                    .first())
        if branding is not None:
            request.domain_workspace = branding.workspace
            request.branding = branding
            if host not in settings.ALLOWED_HOSTS:
                settings.ALLOWED_HOSTS = list(settings.ALLOWED_HOSTS) + [host]
        response = self.get_response(request)
        patch_vary_headers(response, ["Host"])   # ক্যাশ আলাদা রাখে
        return response
