def grade_for(percent, scale):
    """Resolve a percentage to (letter, gpa)."""
    if scale is None or percent is None:
        return ("-", Decimal("0.00"))
    p = Decimal(str(percent))
    for band in scale.bands.all():        # ordering = -min_percent
        if band.min_percent <= p <= band.max_percent:
            return (band.letter, band.gpa)
    return ("-", Decimal("0.00"))

class GradeBand(models.Model):
    letter = models.CharField(max_length=5)
    min_percent = models.DecimalField(max_digits=5, decimal_places=2)
    max_percent = models.DecimalField(max_digits=5, decimal_places=2)
    gpa = models.DecimalField(max_digits=3, decimal_places=2)

def _csv_response(filename, header, rows):
    resp = HttpResponse(content_type="text/csv; charset=utf-8")
    resp["Content-Disposition"] = (
        f'attachment; filename="{filename}"')
    resp.write("\ufeff")          # BOM so Excel opens UTF-8
    writer = csv.writer(resp)
    writer.writerow(header)
    for row in rows:
        writer.writerow(row)
    return resp

        raise PermissionDenied("You are not assigned to this class.")
    return _render_sheet(request, exam, exam.marks..., own=False)
