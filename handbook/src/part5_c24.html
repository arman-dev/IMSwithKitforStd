def test_each_role_is_offered_only_its_own_links(self):
    for email, rules in EXPECTED_LINKS.items():
        with self.subTest(role=email):
            html = self._sidebar_html(email)
            for label in rules["must_show"]:
                self.assertIn(label, html)
            for label in rules["must_hide"]:
                self.assertNotIn(label, html)
