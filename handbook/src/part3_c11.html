def resolve_workspace(request):
    # ১. সেশনে সংরক্ষিত পছন্দ আছে?
    ws_id = request.session.get(
        "active_workspace_id")
    if ws_id and valid(ws_id):
        return membership        # পছন্দ মানুন

    # ২. নেই → সবচেয়ে উঁচু ভূমিকা
    membership = qs.order_by(
        ROLE_PRIORITY,      # ORG_ADMIN আগে
        KIND_PRIORITY,      # প্রতিষ্ঠান আগে
        "workspace__name")  # তারপর নাম
    request.session["active_workspace_id"]
        = membership.workspace_id
    return membership
