def workspace_view(roles=None):
    def deco(view):
        @wraps(view)
        def inner(request, *a, **kw):
            ws = request.workspace          # URL থেকে
            if ws is None:
                raise Http404
            m = membership_for(request.user, ws)
            if m is None:
                raise Http404               # অস্তিত্ব গোপন
            if roles and m.role not in roles:
                raise PermissionDenied       # ৪০৩
            request.membership = m
            return view(request, *a, **kw)
        return inner
    return deco
