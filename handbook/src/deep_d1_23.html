float-এ ০.১ + ০.২ = ০.৩০০০০০০০০০০০০০০৪। এক বছরে হাজার ইনভয়েসে সেটি টাকায় পরিণত হয়। 
@staticmethod
def _next_number():
    year = timezone.localdate().year
    prefix = f"{settings.INVOICE_PREFIX}-{year}-"
    last = (Invoice.objects
            .filter(number__startswith=prefix)
            .order_by("-number")
            .values_list("number", flat=True).first())
    n = int(last.split("-")[-1]) + 1 if last else 1
    return f"{prefix}{n:04d}"
দুর্বলতা: "শেষটি পড়ে এক যোগ" — একই মুহূর্তে দুই ইনভয়েস হলে দুটোই একই নম্বর পায়। number-এ unique=True থাকায় ডেটাবেজ আটকায়, কিন্তু ব্যবহারকারী ৫০০ এরর দেখে।
সঠিক পথ হতো ডেটাবেজ সিকোয়েন্স, অথবা select_for_update() দিয়ে সারি লক করে নম্বর বাড়ানো। এখনো করা হয়নি — এটি প্রকৃত, চিহ্নিত ঋণ।
