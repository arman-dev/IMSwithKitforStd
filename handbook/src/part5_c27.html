def test_expired_trial_goes_read_only_but_keeps_data(self):
    sub = self.sub
    sub.status = Subscription.Status.TRIAL
    sub.trial_ends_on = timezone.localdate() - timedelta(days=1)
    sub.save()
    self.assertFalse(sub.is_writable)                       # AttributeError = লাল
    self.assertEqual(AttendanceRecord.objects.filter(
        workspace=self.ws).count(), 42)                     # ডেটা অটুট

# ---- ২. সবুজ: সবচেয়ে ছোট কোড
@property
def is_writable(self):
    if self.status != self.Status.TRIAL:
        return False
    end = self.trial_ends_on
    return end is None or end >= timezone.localdate()

def test_student_never_sees_unpublished_homework(self):
    hw = Homework.objects.filter(workspace=self.ws).first()
    hw.is_published = False          # শত্রু সারি
    hw.save()

    seen = {h.title for h in
            scoping.visible_homework(self.student, self.ws)}
    self.assertNotIn(hw.title, seen)   # এখন লাল হবে
